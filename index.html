(function () { const CART_STORAGE_KEY = 'caramel_store_cart_v1'; const WHATSAPP_NUMBER = '201112348224'; // غيّر الرقم لو حابب تستقبل أوردرات السلة على رقم مختلف function loadCart() { try { const raw = localStorage.getItem(CART_STORAGE_KEY); return raw ? JSON.parse(raw) : []; } catch (err) { console.error('Error reading cart:', err); return []; } } function saveCart(items) { try { localStorage.setItem(CART_STORAGE_KEY, JSON.stringify(items)); } catch (err) { console.error('Error saving cart:', err); } renderCartBadge(); renderCartModalContent(); } function parsePrice(price) { if (typeof price === 'number' && !isNaN(price)) return price; const cleaned = String(price || '0').replace(/[^\d.]/g, ''); const num = parseFloat(cleaned); return isNaN(num) ? 0 : num; } function formatCurrency(amount) { return `${Number(amount || 0).toLocaleString('ar-EG')} جنيه`; } function showToast(message) { let toast = document.getElementById('caramel-cart-toast'); color: #fff; border: 1px solid #c89666; padding: 10px 18px; border-radius: 999px; font-family: 'Cairo', sans-serif; font-size: 0.88rem; font-weight: 700; z-index: 10000; opacity: 0; transition: all 0.25s ease; }); clearTimeout(toast._timer); toast._timer = setTimeout(() => { toast.style.opacity = '0'; toast.style.transform = 'translateX(-50%) translateY(16px)'; }, 2000); } // دالة إضافة منتج للسلة - يمكن استدعاؤها من أي زرار "أضف للسلة" window.addToCart = function (product) { if (!product || !product.name) return; const cart = loadCart(); const priceNum = parsePrice(product.price); const qtyNum = Math.max(1, parseInt(product.qty || 1, 10)); const id = product.id || `${product.name.trim()}_${priceNum}`; const existingIndex = cart.findIndex(item => item.id === id); if (existingIndex > -1) { cart[existingIndex].qty += qtyNum; } else { cart.push({ id: id, name: String(product.name).trim(), price: priceNum, qty: qtyNum, image: product.image || '' }); } saveCart(cart); showToast(`تمت إضافة "${product.name}" إلى السلة`); }; window.updateCartQty = function (id, delta) { const cart = loadCart(); const idx = cart.findIndex(item => item.id === id); if (idx === -1) return; cart[idx].qty += delta; if (cart[idx].qty <= 0) { cart.splice(idx, 1); } saveCart(cart); }; window.removeFromCart = function (id) { const cart = loadCart().filter(item => item.id !== id); saveCart(cart); }; window.clearCart = function () { saveCart([]); }; window.sendCartToWhatsApp = function () { const cart = loadCart(); if (!cart.length) { showToast('السلة فارغة، أضف منتجات أولاً'); return; } let total = 0; const lines = cart.map((item, index) => { const itemTotal = item.price * item.qty; total += itemTotal; return `${index + 1}. *${item.name}* — الكمية: ${item.qty} × ${item.price} جنيه = *${itemTotal} جنيه*`; }); const message = [ 'مرحباً متجر كارامل، أرغب في تأكيد الطلب التالي:', '', ...lines, '', `الإجمالي الكلي: *${total} جنيه*` ].join('\n'); const url = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(message)}`; window.open(url, '_blank', 'noopener,noreferrer'); }; function renderCartBadge() { const cart = loadCart(); const totalCount = cart.reduce((sum, item) => sum + item.qty, 0); const badge = document.getElementById('caramel-cart-count'); if (badge) { badge.textContent = totalCount; } } function buildCartUI() { if (document.getElementById('caramel-floating-cart-btn')) return; // زر عائم للسلة يظهر في كل الصفحات بدون الحاجة لتعديل الهيدر يدوياً const fab = document.createElement('button'); fab.id = 'caramel-floating-cart-btn'; fab.type = 'button'; fab.setAttribute('aria-label', 'فتح سلة المشتريات'); fab.style.cssText = ` position: fixed; bottom: 20px; left: 20px; z-index: 9990; display: inline-flex; font-size: 0.92rem; font-weight: 900; cursor: pointer; box-shadow: 0 10px 25px rgba(28, 22, 18, 0.35); `; fab.innerHTML = ` السلة 0 `; fab.addEventListener('click', openCartModal); document.body.appendChild(fab); // نافذة السلة المنبثقة const backdrop = document.createElement('div'); backdrop.id = 'caramel-cart-backdrop'; backdrop.style.cssText = ` position: fixed; inset: 0; background: rgba(0, 0, 0, 0.55); backdrop-filter: blur(3px); سلة المشتريات
الإجمالي: 0 جنيه
`; document.body.appendChild(backdrop); backdrop.addEventListener('click', (e) => { if (e.target === backdrop) closeCartModal(); }); document.getElementById('caramel-close-cart-btn').addEventListener('click', closeCartModal); document.getElementById('caramel-send-wa-btn').addEventListener('click', window.sendCartToWhatsApp); document.getElementById('caramel-clear-cart-btn').addEventListener('click', window.clearCart); } function openCartModal() { renderCartModalContent(); const backdrop = document.getElementById('caramel-cart-backdrop'); if (backdrop) backdrop.style.display = 'flex'; } function closeCartModal() { const backdrop = document.getElementById('caramel-cart-backdrop'); if (backdrop) backdrop.style.display = 'none'; } function renderCartModalContent() { const container = document.getElementById('caramel-cart-items'); const totalEl = document.getElementById('caramel-cart-total'); if (!container || !totalEl) return; const cart = loadCart(); if (!cart.length) { container.innerHTML = `

`; totalEl.textContent = formatCurrency(0); return; } let total = 0; container.innerHTML = cart.map(item => { const subtotal = item.price * item.qty; total += subtotal; return `
${item.name}
${item.price} جنيه × ${item.qty} = ${subtotal} جنيه
${item.qty}
`; }).join(''); totalEl.textContent = formatCurrency(total); } document.addEventListener('DOMContentLoaded', () => { buildCartUI(); renderCartBadge(); }); })();
